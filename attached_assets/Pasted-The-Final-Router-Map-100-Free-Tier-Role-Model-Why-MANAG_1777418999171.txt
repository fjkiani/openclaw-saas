// Plumber selection logic:
function selectPlumber(workerResults: WorkerResult[]): string {
  const hasImages = workerResults.some(r => r.hasImageContext);
  const totalTokens = workerResults.reduce((n, r) => n + r.outputTokens, 0);
  if (hasImages) return ROLES.plumber.multimodal;
  if (totalTokens > 4000) return ROLES.plumber.long;
  return ROLES.plumber.short;
}
